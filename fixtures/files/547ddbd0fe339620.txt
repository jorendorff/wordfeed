         That the script loaded proves nothing: it can be blocked, it can return
         no fill, and the unit can be closed or collapsed later -- in each of
         those cases the 105px offset would strand the AnyClip player above an
         empty gap. The unit is viewport-anchored, so look for a visible
         fixed-positioned box of real size inside GumGum's container: the same
         shape of check used for the AnyClip player below, and for the same
         reason -- the vendor's own class names are not ours to depend on.
         The box also has to actually intersect the viewport. A unit closed by
         being translated out of view keeps its size and its fixed position, so
         measuring the box alone would read it as still on screen and strand the
         player over the empty space it left behind. */
      function inScreenIsOnScreen(container) {
        if (!breakpoint.matches || window.getComputedStyle(container).display === 'none') {
          return false;
        }
        const nodes = container.querySelectorAll('*');
        for (let i = 0; i < nodes.length; i++) {
          const style = window.getComputedStyle(nodes[i]);
          if (style.position !== 'fixed' || style.display === 'none' || style.visibility === 'hidden') {
            continue;
          }
          const rect = nodes[i].getBoundingClientRect();
          if (rect.width < MIN_UNIT_WIDTH || rect.height < MIN_UNIT_HEIGHT) {
            continue;
          }
          const onScreen = rect.bottom > 0 && rect.right > 0 &&
            rect.top < window.innerHeight && rect.left < window.innerWidth;
          if (onScreen) {
            return true;
          }
        }
        return false;
      }

      function sync() {
        syncQueued = false;
        const container = window.GUMGUM && window.GUMGUM.container;
        /* A container GumGum has detached counts as gone: it can still be
           referenced by GUMGUM.container after the node leaves the document. */
        if (!container || !document.body.contains(container)) {
          document.body.removeAttribute(INSCREEN_ATTR);
          return;
        }
        if (container !== observedContainer) {
          if (observedContainer) {
            observedContainer.removeEventListener('transitionend', queueSync);
            observedContainer.removeEventListener('animationend', queueSync);
          }
          observedContainer = container;
          /* Hands the node to the stylesheet rule above. Our own attribute, so
             GumGum's inline display is never read or written. */
          container.setAttribute(CONTAINER_ATTR, '');
          /* Catches the unit rendering, collapsing, or being closed later. */
          unitObserver.disconnect();
          unitObserver.observe(container, {
            childList: true,
            subtree: true,
            attributes: true,
            attributeFilter: ['style', 'class']
          });
          /* A CSS close animation only touches style/class when it starts, so
             the observer above fires while the unit is still over the viewport
             and nothing tells us when it has finished leaving. These bubble from
             the unit, so re-check once the movement is actually done. */
          container.addEventListener('transitionend', queueSync);
          container.addEventListener('animationend', queueSync);
        }
        if (inScreenIsOnScreen(container)) {
          document.body.setAttribute(INSCREEN_ATTR, '');
        } else {
          document.body.removeAttribute(INSCREEN_ATTR);
        }
      }

      function queueSync() {
        if (syncQueued) {
          return;
        }
        syncQueued = true;
        window.requestAnimationFrame(sync);
      }

      /* GumGum appends its container to <body> whenever its script finishes, so
         watch for that rather than polling for a fixed window: on a slow
         connection it can arrive long after any timeout we would pick, and if it
         lands while we are below the breakpoint the unit would appear over the
         Adhesion -- the exact collision this is here to prevent.
         This stays connected for the life of the page. unitObserver watches the
         container's subtree and so cannot see the container itself being removed
         from <body>; only this one can, and a close that removes the node has to
         clear the offset. Direct children of <body> only, so it stays cheap. */
      const containerObserver = new MutationObserver(queueSync);
      containerObserver.observe(document.body, { childList: true });
      const unitObserver = new MutationObserver(queueSync);

      breakpoint.addEventListener('change', function () {
        if (breakpoint.matches) {
          loadGumGum();
        }
        queueSync();
      });

      if (breakpoint.matches) {
        loadGumGum();
      }

      /* AnyClip names its own DOM with an `ac-` prefix, not anything we control,
         so we watch for the descendant it switches to position:fixed rather than
         matching a vendor class name that could be renamed under us. It floats
         in place -- the player stays inside our wrapper -- so scoping the scan
         and the observer to the wrapper is enough. */
      /* A data attribute, not a class or an inline style: those two are the
         observer's attributeFilter below, so writing either would re-trigger
         the observer on every mark. */
      const FLOAT_ATTR = 'data-gumgum-anyclip-float';

      const wrapper = document.querySelector('.box-additional-anyclip');
      if (!wrapper) {
        return;
      }

      let scanQueued = false;

      function mark(el) {
        if (window.getComputedStyle(el).position === 'fixed') {
          el.setAttribute(FLOAT_ATTR, '');
        } else if (el.hasAttribute(FLOAT_ATTR)) {
          el.removeAttribute(FLOAT_ATTR);
        }
      }

      function scan() {
        scanQueued = false;
        const descendants = wrapper.querySelectorAll('*');
        for (let i = 0; i < descendants.length; i++) {
          mark(descendants[i]);
        }
      }

      function queueScan() {
        if (scanQueued) {
          return;
        }
        scanQueued = true;
        window.requestAnimationFrame(scan);
      }

      const observer = new MutationObserver(queueScan);
      observer.observe(wrapper, {
        childList: true,
        subtree: true,
        attributes: true,
        attributeFilter: ['style', 'class']
      });
      queueScan();
    })();
  </script>
